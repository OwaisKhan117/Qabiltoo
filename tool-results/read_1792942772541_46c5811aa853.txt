    63→
    64→function StatusChip({ status }: { status: InterestWithProfile["status"] }) {
    65→  if (status === "accepted") {
    66→    return (
    67→      <span className="inline-flex items-center gap-1 rounded-full bg-emerald-100 px-2.5 py-0.5 text-[11px] font-semibold text-emerald-800 dark:bg-emerald-900/50 dark:text-emerald-300">
    68→        <HeartHandshake className="h-3 w-3" aria-hidden />
    69→        Accepted
    70→      </span>
    71→    );
    72→  }
    73→  if (status === "declined") {
    74→    return (
    75→      <span className="inline-flex items-center gap-1 rounded-full bg-muted px-2.5 py-0.5 text-[11px] font-medium text-muted-foreground">
    76→        <UserRoundX className="h-3 w-3" aria-hidden />
    77→        Declined
    78→      </span>
    79→    );
    80→  }
    81→  return (
    82→    <span className="inline-flex items-center gap-1 rounded-full bg-gold-100 px-2.5 py-0.5 text-[11px] font-semibold text-gold-800 dark:bg-gold-500/15 dark:text-gold-300">
    83→      <Sparkles className="h-3 w-3" aria-hidden />
    84→      Pending
    85→    </span>
    86→  );
    87→}

    89→/* ── Main view ────────────────────────────────────────────── */
    90→
    91→export function MatchesView({
    92→  navigate,
    93→  onViewProfile,
    94→}: {
    95→  navigate: (v: View) => void;
    96→  onViewProfile: (id: string) => void;
    97→}) {
    98→  const mounted = useMounted();
    99→  const user = useAuthStore((s) => s.user);
   100→  const savedIdsRaw = useAuthStore((s) => s.savedIds);
   101→  const toggleSaved = useAuthStore((s) => s.toggleSaved);
   102→  const queryClient = useQueryClient();
   103→  const [revealPair, setRevealPair] = useState<Profile | null>(null);
   104→
   105→  const authed = mounted && !!user;
   106→
   107→  /* ── Data ── */
   108→  const { data: interests, isLoading } = useQuery({
   109→    queryKey: ["interests", user?.id],
   110→    queryFn: async () => {
   111→      const res = await fetch(`/api/interests?profileId=${user!.id}`);
   112→      if (!res.ok) throw new Error("Failed to fetch proposals");
   113→      return (await res.json()) as { sent: InterestWithProfile[]; received: InterestWithProfile[] };
   114→    },
   115→    enabled: authed,
   116→    staleTime: 10_000,
   117→  });
   118→
   119→  const { data: viewsData } = useQuery({
   120→    queryKey: ["views", user?.id],
   121→    queryFn: async () => {
   122→      const res = await fetch(`/api/views?profileId=${user!.id}&tier=${encodeURIComponent(user!.tier)}`);
   123→      if (!res.ok) throw new Error("Failed to fetch views");
   124→      return (await res.json()) as { viewers: ViewerItem[]; total: number };
   125→    },
   126→    enabled: authed,
   127→    staleTime: 15_000,
   128→  });
   129→
   130→  const savedIds = useMemo(
   131→    () => (authed ? savedIdsRaw.filter((id) => id !== user?.id) : []),
   132→    [authed, savedIdsRaw, user?.id]
   133→  );
   134→
   135→  const savedQueries = useQueries({
   136→    queries: savedIds.map((id) => ({
   137→      queryKey: ["profile", id],
   138→      queryFn: async () => {
   139→        const res = await fetch(`/api/profiles/${id}`);
   140→        if (!res.ok) return { profile: null };
   141→        return (await res.json()) as { profile: Profile | null };
   142→      },
   143→      staleTime: 60_000,
   144→      enabled: authed,
   145→    })),
   146→  });
   147→
   148→  const savedProfiles = useMemo(
   149→    () =>
   150→      savedQueries
   151→        .map((q) => q.data?.profile ?? null)
   152→        .filter((p): p is Profile => !!p),
   153→    [savedQueries]
   154→  );
   155→
   156→  /* ── Mutual matches (sent accepted ∩ received accepted) ── */
   157→  const mutualIds = useMemo(() => {
   158→    const sentAccepted = new Set(
   159→      interests?.sent.filter((i) => i.status === "accepted").map((i) => i.toId) ?? []
   160→    );
   161→    const receivedAccepted = new Set(
   162→      interests?.received.filter((i) => i.status === "accepted").map((i) => i.fromId) ?? []
   163→    );
   164→    return new Set([...sentAccepted].filter((id) => receivedAccepted.has(id)));
   165→  }, [interests]);
   166→
   167→  const mutualProfiles = useMemo(() => {
   168→    if (!interests) return [];
   169→    return interests.received
   170→      .filter((i) => mutualIds.has(i.fromId))
   171→      .map((i) => i.from as Profile);
   172→  }, [interests, mutualIds]);
   173→
   174→  /* ── Accept / decline (optimistic) ── */
   175→  const respond = useMutation({
   176→    mutationFn: async ({ id, status }: { id: string; status: "accepted" | "declined" }) => {
   177→      const res = await fetch(`/api/interests/${id}`, {
   178→        method: "PATCH",
   179→        headers: { "Content-Type": "application/json" },
   180→        body: JSON.stringify({ status, profileId: user!.id }),
   181→      });
   182→      if (!res.ok) {
   183→        const data = (await res.json().catch(() => ({}))) as { error?: string };
   184→        throw new Error(data.error ?? "Could not respond to the proposal");
   185→      }
   186→      return res.json();
   187→    },
   188→    onMutate: async ({ id, status }) => {
   189→      await queryClient.cancelQueries({ queryKey: ["interests", user!.id] });
   190→      const key = ["interests", user!.id] as const;
   191→      const previous = queryClient.getQueryData<{
   192→        sent: InterestWithProfile[];
   193→        received: InterestWithProfile[];
   194→      }>(key);
   195→      queryClient.setQueryData<{ sent: InterestWithProfile[]; received: InterestWithProfile[] }>(
   196→        key,
   197→        (old) =>
   198→          old
   199→            ? {
   200→                sent: old.sent,
   201→                received: old.received.map((i) => (i.id === id ? { ...i, status } : i)),
   202→              }
   203→            : old
   204→      );
   205→      return { previous, key };
   206→    },
   207→    onError: (_err, _vars, ctx) => {
   208→      if (ctx?.previous) queryClient.setQueryData(ctx.key, ctx.previous);
   209→      toast({
   210→        title: "Something went wrong",
   211→        description: "Please try again in a moment.",
   212→        variant: "destructive",
   213→      });
   214→    },
   215→    onSuccess: (_data, vars) => {
   216→      queryClient.invalidateQueries({ queryKey: ["interests", user!.id] });
   217→      queryClient.invalidateQueries({ queryKey: ["notifications"] });
   218→      if (vars.status === "accepted") {
   219→        const interest = interests?.received.find((i) => i.id === vars.id);
   220→        if (interest && mutualIds.has(interest.fromId)) {
   221→          setRevealPair(interest.from as Profile);
   222→        }
   223→        toast({
   224→          title: "Proposal accepted — MashaAllah",
   225→          description: mutualIds.has(interest?.fromId ?? "")
   226→            ? "It's a match. A respectful conversation has opened."
   227→            : "Chat unlocks when you've both accepted.",
   228→        });
   229→      } else {
   230→        toast({
   231→          title: "Handled with grace",
   232→          description: "The proposal has been respectfully declined.",
   233→        });
   234→      }
   235→    },
   236→  });
   237→
   238→  /* ── Loading (pre-hydration / fetching) ── */
   239→  if (!mounted) {
   240→    return (
   241→      <div className="mx-auto max-w-6xl px-4 py-16">
   242→        <div className="flex flex-col items-center gap-4">
   243→          <Skeleton className="h-8 w-64" />
   244→          <Skeleton className="h-4 w-40" />
   245→        </div>
   246→        <div className="mt-10 grid gap-5 sm:grid-cols-2 lg:grid-cols-3">
   247→          {[0, 1, 2].map((i) => (
   248→            <Skeleton key={i} className="h-64 rounded-3xl" />
   249→          ))}
   250→        </div>
   251→      </div>
   252→    );
   253→  }
   254→
   255→  if (!user) {
   256→    return (
   257→      <SignInState
   258→        title="Sign in to view your proposals"
   259→        hint="Your received proposals, sent proposals and profile visitors are kept private in your account."
   260→      />
   261→    );
   262→  }
   263→
   264→  const received = interests?.received ?? [];
   265→  const sent = interests?.sent ?? [];
   266→  const pendingReceived = received.filter((i) => i.status === "pending");
   267→
   268→  return (
   269→    <div className="mx-auto max-w-6xl px-4 pb-20 pt-10 sm:px-6">
   270→      <SectionHeading
   271→        arabic="الطلبات"
   272→        eyebrow="Proposals & Matches"
   273→        title="Every proposal, handled with dignity"
   274→        description="Review proposals with your family, respond with grace, and let conversation open only when hearts meet on both sides."
   275→      />
   276→
   277→      {/* ── Your matches strip ── */}
   278→      {mutualProfiles.length > 0 && (
   279→        <section aria-label="Your matches" className="mb-8">
   280→          <div className="glass-card rounded-3xl border border-gold-500/25 bg-gradient-to-r from-gold-50 via-background to-emerald-50 p-5 dark:from-gold-500/5 dark:via-transparent dark:to-emerald-900/20">
   281→            <div className="mb-4 flex items-center gap-2">
   282→              <span className="flex h-8 w-8 items-center justify-center rounded-full bg-gold-500/15 text-gold-600 dark:text-gold-400">
   283→                <HeartHandshake className="h-4 w-4" aria-hidden />
   284→              </span>
   285→              <h2 className="font-display text-lg font-semibold">Your Matches</h2>
   286→              <Badge className="bg-gold-500/15 text-[11px] text-gold-800 hover:bg-gold-500/15 dark:text-gold-300">
   287→                {mutualProfiles.length} mutual
   288→              </Badge>
   289→            </div>
   290→            <div className="flex gap-4 overflow-x-auto pb-1 scrollbar-thin">
   291→              {mutualProfiles.map((p) => (
   292→                <motion.button
   293→                  key={p.id}
   294→                  whileHover={{ y: -3 }}
   295→                  whileTap={{ scale: 0.97 }}
   296→                  onClick={() => setRevealPair(p)}
   297→                  className="group flex w-36 shrink-0 flex-col items-center gap-2 rounded-2xl border border-gold-500/20 bg-background/70 p-3 text-center shadow-sm transition-colors hover:border-gold-500/45"
   298→                  aria-label={`Open match with ${displayName(p.fullName)}`}
   299→                >
   300→                  <RoundPhoto
   301→                    profile={p}
   302→                    alt={p.fullName}
   303→                    className="h-16 w-16"
   304→                    sizes="64px"
   305→                  />
   306→                  <span className="text-sm font-semibold leading-tight">
   307→                    {displayName(p.fullName)}
   308→                  </span>
   309→                  <span className="flex items-center gap-1 text-[11px] text-gold-700 dark:text-gold-400">
   310→                    <MessageCircle className="h-3 w-3" aria-hidden />
   311→                    Open chat
   312→                  </span>
   313→                </motion.button>
   314→              ))}
   315→            </div>
   316→          </div>
   317→        </section>
   318→      )}
   319→
   320→      {/* ── Tabs ── */}
   321→      <Tabs defaultValue="received">
   322→        <TabsList className="mb-6 flex h-auto w-full flex-wrap justify-start gap-1 rounded-2xl bg-muted/60 p-1.5 sm:inline-flex sm:w-auto">
   323→          <TabsTrigger value="received" className="gap-1.5 rounded-xl px-3 py-2 text-sm">
   324→            <Inbox className="h-4 w-4" aria-hidden />
   325→            Received
   326→            {pendingReceived.length > 0 && (
   327→              <span className="ml-0.5 flex h-5 min-w-5 items-center justify-center rounded-full bg-gold-500 px-1 text-[10px] font-bold text-emerald-950">
   328→                {pendingReceived.length}
   329→              </span>
   330→            )}
   331→          </TabsTrigger>
   332→          <TabsTrigger value="sent" className="gap-1.5 rounded-xl px-3 py-2 text-sm">
   333→            <MailOpen className="h-4 w-4" aria-hidden />
   334→            Sent
   335→            {sent.length > 0 && (
   336→              <span className="text-xs text-muted-foreground">{sent.length}</span>
   337→            )}
   338→          </TabsTrigger>
   339→          <TabsTrigger value="views" className="gap-1.5 rounded-xl px-3 py-2 text-sm">
   340→            <Eye className="h-4 w-4" aria-hidden />
   341→            Who Viewed Me
   342→            {(viewsData?.total ?? 0) > 0 && (
   343→              <span className="text-xs text-muted-foreground">{viewsData?.total}</span>
   344→            )}
   345→          </TabsTrigger>
   346→          <TabsTrigger value="saved" className="gap-1.5 rounded-xl px-3 py-2 text-sm">
   347→            <Gem className="h-4 w-4" aria-hidden />
   348→            Saved
   349→            {savedIds.length > 0 && (
   350→              <span className="text-xs text-muted-foreground">{savedIds.length}</span>
   351→            )}
   352→          </TabsTrigger>
   353→        </TabsList>
   354→
   355→        {/* ═══ RECEIVED ═══ */}
   356→        <TabsContent value="received">
   357→          {isLoading ? (
   358→            <div className="grid gap-5 sm:grid-cols-2 lg:grid-cols-3">
   359→              {[0, 1, 2].map((i) => (
   360→                <Skeleton key={i} className="h-72 rounded-3xl" />
   361→              ))}
   362→            </div>
   363→          ) : received.length === 0 ? (
   364→            <EmptyState
   365→              icon={<Inbox className="h-6 w-6" aria-hidden />}
   366→              title="No proposals yet"
   367→              body="When someone sends a proposal with sincere intentions, it will appear here for you and your family to review."
   368→              action={
   369→                <Button variant="outline" onClick={() => navigate({ name: "browse" })}>
   370→                  Discover Profiles
   371→                </Button>
   372→              }
   373→            />
   374→          ) : (
   375→            <div className="grid gap-5 md:grid-cols-2 lg:grid-cols-3">
   376→              {received.map((interest) => {
   377→                const sender = interest.from as Profile;
   378→                const isMutual = mutualIds.has(sender.id);
   379→                const handled = interest.status !== "pending";
   380→                return (
   381→                  <ProposalCardShell key={interest.id} className="flex flex-col">
   382→                    <div className="flex items-start gap-4">
   383→                      <ArchPhoto
   384→                        profile={sender}
   385→                        alt={sender.fullName}
   386→                        className="h-24 w-20"
   387→                        sizes="80px"
   388→                      />
   389→                      <div className="min-w-0 flex-1">
   390→                        <button
   391→                          onClick={() => onViewProfile(sender.id)}
   392→                          className="block max-w-full truncate text-left font-display text-lg font-semibold leading-snug hover:text-gold-700 dark:hover:text-gold-400"
   393→                        >
   394→                          {displayName(sender.fullName)}
   395→                        </button>
   396→                        <p className="truncate text-sm text-muted-foreground">
   397→                          {sender.age} · {sender.city}
   398→                          {sender.country !== "India" ? `, ${sender.country}` : ""}
   399→                        </p>
   400→                        <p className="truncate text-sm text-muted-foreground">{sender.profession}</p>
   401→                        <div className="mt-1.5">
   402→                          <VerificationBadge tier={sender.verificationTier} />
   403→                        </div>
   404→                      </div>
   405→                      <span className="shrink-0 text-[11px] text-muted-foreground">
   406→                        {timeAgo(interest.createdAt)}
   407→                      </span>
   408→                    </div>
   409→
   410→                    {interest.message && (
   411→                      <blockquote className="mt-4 rounded-2xl border-l-2 border-gold-500/60 bg-gold-50/60 px-4 py-3 text-sm italic leading-relaxed text-foreground/85 dark:bg-gold-500/5">
   412→                        &ldquo;{interest.message}&rdquo;
   413→                      </blockquote>
   414→                    )}
   415→
   416→                    <div className="mt-4 flex flex-1 items-end">
   417→                      {handled ? (
   418→                        <div className="flex w-full items-center justify-between gap-2">
   419→                          <StatusChip status={interest.status} />
   420→                          {isMutual && (
   421→                            <Button
   422→                              size="sm"
   423→                              onClick={() => navigate({ name: "chat", withId: sender.id })}
   424→                              className="gap-1.5 bg-emerald-700 text-white hover:bg-emerald-600"
   425→                            >
   426→                              <MessageCircle className="h-3.5 w-3.5" aria-hidden />
   427→                              Conversation
   428→                            </Button>
   429→                          )}
   430→                        </div>
   431→                      ) : (
   432→                        <div className="flex w-full gap-2">
   433→                          <Button
   434→                            className="btn-gold-shimmer flex-1 bg-gold-500 font-semibold text-emerald-950 hover:bg-gold-400"
   435→                            disabled={respond.isPending}
   436→                            onClick={() => respond.mutate({ id: interest.id, status: "accepted" })}
   437→                          >
   438→                            Accept
   439→                          </Button>
   440→                          <Button
   441→                            variant="outline"
   442→                            className="flex-1"
   443→                            disabled={respond.isPending}
   444→                            onClick={() => respond.mutate({ id: interest.id, status: "declined" })}
   445→                          >
   446→                            Decline
   447→                          </Button>
   448→                        </div>
   449→                      )}
   450→                    </div>
   451→                  </ProposalCardShell>
   452→                );
   453→              })}
   454→            </div>
   455→          )}
   456→        </TabsContent>
   457→
   458→        {/* ═══ SENT ═══ */}
   459→        <TabsContent value="sent">
   460→          {sent.length === 0 ? (
   461→            <EmptyState
   462→              icon={<MailOpen className="h-6 w-6" aria-hidden />}
   463→              title="No proposals sent yet"
   464→              body="When you send a respectful proposal, you can follow its journey here — from pending to accepted, Insha'Allah."
   465→              action={
   466→                <Button variant="outline" onClick={() => navigate({ name: "browse" })}>
   467→                  Discover Profiles
   468→                </Button>
   469→              }
   470→            />
   471→          ) : (
   472→            <div className="grid gap-5 md:grid-cols-2 lg:grid-cols-3">
   473→              {sent.map((interest) => {
   474→                const recipient = interest.to as Profile;
   475→                const isMutual = mutualIds.has(recipient.id);
   476→                return (
   477→                  <ProposalCardShell key={interest.id} className="flex flex-col">
   478→                    <div className="flex items-start gap-4">
   479→                      {interest.status === "accepted" ? (
   480→                        <ArchPhoto
   481→                          profile={recipient}
   482→                          alt={recipient.fullName}
   483→                          className="h-24 w-20"
   484→                          sizes="80px"
   485→                        />
   486→                      ) : (
   487→                        <BlurredPhoto
   488→                          src={recipient.image}
   489→                          alt={recipient.fullName}
   490→                          className="h-24 w-20 shrink-0 rounded-t-[3.5rem] rounded-b-2xl ring-1 ring-gold-500/25"
   491→                          overlayLabel="Revealed when accepted"
   492→                          onClick={() => onViewProfile(recipient.id)}
   493→                        />
   494→                      )}
   495→                      <div className="min-w-0 flex-1">
   496→                        <button
   497→                          onClick={() => onViewProfile(recipient.id)}
   498→                          className="block max-w-full truncate text-left font-display text-lg font-semibold leading-snug hover:text-gold-700 dark:hover:text-gold-400"
   499→                        >
   500→                          {displayName(recipient.fullName)}
   501→                        </button>
   502→                        <p className="truncate text-sm text-muted-foreground">
   503→                          {recipient.age} · {recipient.city}
   504→                          {recipient.country !== "India" ? `, ${recipient.country}` : ""}
   505→                        </p>
   506→                        <p className="truncate text-sm text-muted-foreground">{recipient.profession}</p>
   507→                        <div className="mt-1.5">
   508→                          <VerificationBadge tier={recipient.verificationTier} />
   509→                        </div>
   510→                      </div>
   511→                      <span className="shrink-0 text-[11px] text-muted-foreground">
   512→                        {timeAgo(interest.createdAt)}
   513→                      </span>
   514→                    </div>
   515→
   516→                    {interest.message && (
   517→                      <blockquote className="mt-4 rounded-2xl border-l-2 border-emerald-600/40 bg-emerald-50/60 px-4 py-3 text-sm italic leading-relaxed text-foreground/80 dark:bg-emerald-900/15">
   518→                        &ldquo;{interest.message}&rdquo;
   519→                      </blockquote>
   520→                    )}
   521→
   522→                    <div className="mt-4 flex items-center justify-between gap-2">
   523→                      <StatusChip status={interest.status} />
   524→                      {isMutual ? (
   525→                        <Button
   526→                          size="sm"
   527→                          onClick={() => navigate({ name: "chat", withId: recipient.id })}
   528→                          className="gap-1.5 bg-emerald-700 text-white hover:bg-emerald-600"
   529→                        >
   530→                          <MessageCircle className="h-3.5 w-3.5" aria-hidden />
   531→                          Message
   532→                        </Button>
   533→                      ) : interest.status === "accepted" ? (
   534→                        <Button
   535→                          size="sm"
   536→                          variant="outline"
   537→                          onClick={() => navigate({ name: "chat", withId: recipient.id })}
   538→                          className="gap-1.5"
   539→                        >
   540→                          <MessageCircle className="h-3.5 w-3.5" aria-hidden />
   541→                          Message
   542→                        </Button>
   543→                      ) : null}
   544→                    </div>
   545→                  </ProposalCardShell>
   546→                );
   547→              })}
   548→            </div>
   549→          )}
   550→        </TabsContent>
   551→
   552→        {/* ═══ WHO VIEWED ME ═══ */}
   553→        <TabsContent value="views">
   554→          {!viewsData ? (
   555→            <div className="grid gap-5 sm:grid-cols-2 lg:grid-cols-3">
   556→              {[0, 1, 2].map((i) => (
   557→                <Skeleton key={i} className="h-44 rounded-3xl" />
   558→              ))}
   559→            </div>
   560→          ) : viewsData.viewers.length === 0 ? (
   561→            <EmptyState
   562→              icon={<Eye className="h-6 w-6" aria-hidden />}
   563→              title="No profile visits yet"
   564→              body="As your profile reaches more members, you'll see who stopped by — a gentle signal of interest."
   565→            />
   566→          ) : (
   567→            <>
   568→              <p className="mb-4 text-sm text-muted-foreground">
   569→                {viewsData.total} {viewsData.total === 1 ? "member has" : "members have"} viewed your
   570→                profile recently.
   571→                {user.tier !== "Gold" && user.tier !== "Platinum" && " Upgrade to reveal every visitor."}

   586→                          <span className="block truncate font-display text-base font-semibold">
   587→                            {displayName(v.viewer.fullName)}
   588→                          </span>
   589→                          <span className="block truncate text-sm text-muted-foreground">
   590→                            {v.viewer.city}
   591→                            {v.viewer.country !== "India" ? `, ${v.viewer.country}` : ""}
   592→                          </span>
   593→                          <span className="text-[11px] text-muted-foreground/80">
   594→                            Viewed {timeAgo(v.viewedAt)} ago
   595→                          </span>
   596→                        </span>
   597→                      </button>
   598→                    </ProposalCardShell>
   599→                  )
   600→                )}
   601→              </div>
   602→            </>
   603→          )}
   604→        </TabsContent>
   605→
   606→        {/* ═══ SAVED ═══ */}
   607→        <TabsContent value="saved">
   608→          {savedIds.length === 0 ? (
   609→            <EmptyState
   610→              icon={<Gem className="h-6 w-6" aria-hidden />}
   611→              title="Nothing saved yet"
   612→              body="Tap the bookmark on any profile to keep it here for thoughtful review with your family."
   613→              action={
   614→                <Button variant="outline" onClick={() => navigate({ name: "browse" })}>
   615→                  Discover Profiles
   616→                </Button>
   617→              }
   618→            />
   619→          ) : (
   620→            <div className="grid gap-5 sm:grid-cols-2 lg:grid-cols-3">
   621→              {savedProfiles.map((p) => (
   622→                <ProposalCardShell key={p.id} className="flex flex-col">
   623→                  <div className="flex items-start gap-4">
   624→                    <BlurredPhoto
   625→                      src={p.image}
   626→                      alt={p.fullName}
   627→                      className="h-24 w-20 shrink-0 rounded-t-[3.5rem] rounded-b-2xl ring-1 ring-gold-500/25"
   628→                      overlayLabel="Photos stay private"
   629→                      onClick={() => onViewProfile(p.id)}
   630→                    />
   631→                    <div className="min-w-0 flex-1">
   632→                      <button
   633→                        onClick={() => onViewProfile(p.id)}
   634→                        className="block max-w-full truncate text-left font-display text-lg font-semibold leading-snug hover:text-gold-700 dark:hover:text-gold-400"
   635→                      >
   636→                        {displayName(p.fullName)}
   637→                      </button>
   638→                      <p className="truncate text-sm text-muted-foreground">
   639→                        {p.age} · {p.city}
   640→                        {p.country !== "India" ? `, ${p.country}` : ""}
   641→                      </p>
   642→                      <p className="truncate text-sm text-muted-foreground">{p.profession}</p>
   643→                      <div className="mt-1.5">
   644→                        <VerificationBadge tier={p.verificationTier} />
   645→                      </div>
   646→                    </div>
   647→                  </div>
   648→                  <div className="mt-4 flex items-center justify-between">
   649→                    <Button
   650→                      size="sm"
   651→                      variant="outline"
   652→                      onClick={() => onViewProfile(p.id)}
   653→                      className="gap-1.5"
   654→                    >
   655→                      View Profile
   656→                    </Button>
   657→                    <Button
   658→                      size="sm"
   659→                      variant="ghost"
   660→                      onClick={() => {
   661→                        toggleSaved(p.id);
   662→                        toast({ title: "Removed from saved", description: `${displayName(p.fullName)} is no longer in your saved list.` });
   663→                      }}
   664→                      className="gap-1.5 text-muted-foreground hover:text-rose-700 dark:hover:text-rose-400"
   665→                      aria-label={`Remove ${displayName(p.fullName)} from saved`}
   666→                    >
   667→                      <BookmarkX className="h-4 w-4" aria-hidden />
   668→                      Unsave
   669→                    </Button>
   670→                  </div>
   671→                </ProposalCardShell>
   672→              ))}
   673→            </div>
   674→          )}
   675→        </TabsContent>
   676→      </Tabs>
   677→
   678→      {/* ── Mutual match reveal ── */}
   679→      <MatchReveal
   680→        open={!!revealPair}
   681→        onOpenChange={(open) => {
   682→          if (!open) setRevealPair(null);
   683→        }}
   684→        user={user}
   685→        other={revealPair ?? user}
   686→        onOpenChat={() => {
   687→          const id = revealPair?.id;
   688→          setRevealPair(null);
   689→          if (id) navigate({ name: "chat", withId: id });
   690→        }}
   691→        onContinue={() => {
   692→          setRevealPair(null);
   693→          navigate({ name: "browse" });
   694→        }}
   695→      />
   696→    </div>
   697→  );
   698→}
