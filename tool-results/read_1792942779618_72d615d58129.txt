    90→    90→
    91→    91→export function MatchesView({
    92→    92→  navigate,
    93→    93→  onViewProfile,
    94→    94→}: {
    95→    95→  navigate: (v: View) => void;
    96→    96→  onViewProfile: (id: string) => void;
    97→    97→}) {
    98→    98→  const mounted = useMounted();
    99→    99→  const user = useAuthStore((s) => s.user);
   100→   100→  const savedIdsRaw = useAuthStore((s) => s.savedIds);
   101→   101→  const toggleSaved = useAuthStore((s) => s.toggleSaved);
   102→   102→  const queryClient = useQueryClient();
   103→   103→  const [revealPair, setRevealPair] = useState<Profile | null>(null);
   104→   104→
   105→   105→  const authed = mounted && !!user;
   106→   106→
   107→   107→  /* ── Data ── */
   108→   108→  const { data: interests, isLoading } = useQuery({
   109→   109→    queryKey: ["interests", user?.id],
   110→   110→    queryFn: async () => {
   111→   111→      const res = await fetch(`/api/interests?profileId=${user!.id}`);
   112→   112→      if (!res.ok) throw new Error("Failed to fetch proposals");
   113→   113→      return (await res.json()) as { sent: InterestWithProfile[]; received: InterestWithProfile[] };
   114→   114→    },
   115→   115→    enabled: authed,
   116→   116→    staleTime: 10_000,
   117→   117→  });
   118→   118→
   119→   119→  const { data: viewsData } = useQuery({
   120→   120→    queryKey: ["views", user?.id],
   121→   121→    queryFn: async () => {
   122→   122→      const res = await fetch(`/api/views?profileId=${user!.id}&tier=${encodeURIComponent(user!.tier)}`);
   123→   123→      if (!res.ok) throw new Error("Failed to fetch views");
   124→   124→      return (await res.json()) as { viewers: ViewerItem[]; total: number };
   125→   125→    },
   126→   126→    enabled: authed,
   127→   127→    staleTime: 15_000,
   128→   128→  });
   129→   129→
   130→   130→  const savedIds = useMemo(
   131→   131→    () => (authed ? savedIdsRaw.filter((id) => id !== user?.id) : []),
   132→   132→    [authed, savedIdsRaw, user?.id]
   133→   133→  );
   134→   134→
   135→   135→  const savedQueries = useQueries({
   136→   136→    queries: savedIds.map((id) => ({
   137→   137→      queryKey: ["profile", id],
   138→   138→      queryFn: async () => {
   139→   139→        const res = await fetch(`/api/profiles/${id}`);
   140→   140→        if (!res.ok) return { profile: null };
   141→   141→        return (await res.json()) as { profile: Profile | null };
   142→   142→      },
   143→   143→      staleTime: 60_000,
   144→   144→      enabled: authed,
   145→   145→    })),
   146→   146→  });
   147→   147→
   148→   148→  const savedProfiles = useMemo(
   149→   149→    () =>
   150→   150→      savedQueries
   151→   151→        .map((q) => q.data?.profile ?? null)
   152→   152→        .filter((p): p is Profile => !!p),
   153→   153→    [savedQueries]
   154→   154→  );
   155→   155→
   156→   156→  /* ── Mutual matches (sent accepted ∩ received accepted) ── */
   157→   157→  const mutualIds = useMemo(() => {
   158→   158→    const sentAccepted = new Set(
   159→   159→      interests?.sent.filter((i) => i.status === "accepted").map((i) => i.toId) ?? []
   160→   160→    );
   161→   161→    const receivedAccepted = new Set(
   162→   162→      interests?.received.filter((i) => i.status === "accepted").map((i) => i.fromId) ?? []
   163→   163→    );
   164→   164→    return new Set([...sentAccepted].filter((id) => receivedAccepted.has(id)));
   165→   165→  }, [interests]);
   166→   166→
   167→   167→  const mutualProfiles = useMemo(() => {
   168→   168→    if (!interests) return [];
   169→   169→    return interests.received
   170→   170→      .filter((i) => mutualIds.has(i.fromId))
   171→   171→      .map((i) => i.from as Profile);
   172→   172→  }, [interests, mutualIds]);
   173→   173→
   174→   174→  /* ── Accept / decline (optimistic) ── */
   175→   175→  const respond = useMutation({
   176→   176→    mutationFn: async ({ id, status }: { id: string; status: "accepted" | "declined" }) => {
   177→   177→      const res = await fetch(`/api/interests/${id}`, {
   178→   178→        method: "PATCH",
   179→   179→        headers: { "Content-Type": "application/json" },
   180→   180→        body: JSON.stringify({ status, profileId: user!.id }),
   181→   181→      });
   182→   182→      if (!res.ok) {
   183→   183→        const data = (await res.json().catch(() => ({}))) as { error?: string };
   184→   184→        throw new Error(data.error ?? "Could not respond to the proposal");
   185→   185→      }
   186→   186→      return res.json();
   187→   187→    },
   188→   188→    onMutate: async ({ id, status }) => {
   189→   189→      await queryClient.cancelQueries({ queryKey: ["interests", user!.id] });
   190→   190→      const key = ["interests", user!.id] as const;
   191→   191→      const previous = queryClient.getQueryData<{
   192→   192→        sent: InterestWithProfile[];
   193→   193→        received: InterestWithProfile[];
   194→   194→      }>(key);
   195→   195→      queryClient.setQueryData<{ sent: InterestWithProfile[]; received: InterestWithProfile[] }>(
   196→   196→        key,
   197→   197→        (old) =>
   198→   198→          old
   199→   199→            ? {
   200→   200→                sent: old.sent,
   201→   201→                received: old.received.map((i) => (i.id === id ? { ...i, status } : i)),
   202→   202→              }
   203→   203→            : old
   204→   204→      );
   205→   205→      return { previous, key };
   206→   206→    },
   207→   207→    onError: (_err, _vars, ctx) => {
   208→   208→      if (ctx?.previous) queryClient.setQueryData(ctx.key, ctx.previous);
   209→   209→      toast({
   210→   210→        title: "Something went wrong",
   211→   211→        description: "Please try again in a moment.",
   212→   212→        variant: "destructive",
   213→   213→      });
   214→   214→    },
   215→   215→    onSuccess: (_data, vars) => {
   216→   216→      queryClient.invalidateQueries({ queryKey: ["interests", user!.id] });
   217→   217→      queryClient.invalidateQueries({ queryKey: ["notifications"] });
   218→   218→      if (vars.status === "accepted") {
   219→   219→        const interest = interests?.received.find((i) => i.id === vars.id);
   220→   220→        if (interest && mutualIds.has(interest.fromId)) {
   221→   221→          setRevealPair(interest.from as Profile);
   222→   222→        }
   223→   223→        toast({
   224→   224→          title: "Proposal accepted — MashaAllah",
   225→   225→          description: mutualIds.has(interest?.fromId ?? "")
   226→   226→            ? "It's a match. A respectful conversation has opened."
   227→   227→            : "Chat unlocks when you've both accepted.",
   228→   228→        });
   229→   229→      } else {
   230→   230→        toast({
   231→   231→          title: "Handled with grace",
   232→   232→          description: "The proposal has been respectfully declined.",
   233→   233→        });
   234→   234→      }
   235→   235→    },
   236→   236→  });
   237→   237→
   238→   238→  /* ── Loading (pre-hydration / fetching) ── */
   239→   239→  if (!mounted) {
   240→   240→    return (
   241→   241→      <div className="mx-auto max-w-6xl px-4 py-16">
   242→   242→        <div className="flex flex-col items-center gap-4">
   243→   243→          <Skeleton className="h-8 w-64" />
   244→   244→          <Skeleton className="h-4 w-40" />
   245→   245→        </div>
   246→   246→        <div className="mt-10 grid gap-5 sm:grid-cols-2 lg:grid-cols-3">
   247→   247→          {[0, 1, 2].map((i) => (
   248→   248→            <Skeleton key={i} className="h-64 rounded-3xl" />
   249→   249→          ))}
   250→   250→        </div>
   251→   251→      </div>
   252→   252→    );
   253→   253→  }
   254→   254→
   255→   255→  if (!user) {
   256→   256→    return (
   257→   257→      <SignInState
   258→   258→        title="Sign in to view your proposals"
   259→   259→        hint="Your received proposals, sent proposals and profile visitors are kept private in your account."
   260→   260→      />
   261→   261→    );
   262→   262→  }
   263→   263→
   264→   264→  const received = interests?.received ?? [];
   265→   265→  const sent = interests?.sent ?? [];
   266→   266→  const pendingReceived = received.filter((i) => i.status === "pending");
   267→   267→
   268→   268→  return (
   269→   269→    <div className="mx-auto max-w-6xl px-4 pb-20 pt-10 sm:px-6">
   270→   270→      <SectionHeading
   271→   271→        arabic="الطلبات"
   272→   272→        eyebrow="Proposals & Matches"
   273→   273→        title="Every proposal, handled with dignity"
   274→   274→        description="Review proposals with your family, respond with grace, and let conversation open only when hearts meet on both sides."
   275→   275→      />
   276→   276→
   277→   277→      {/* ── Your matches strip ── */}
   278→   278→      {mutualProfiles.length > 0 && (
   279→   279→        <section aria-label="Your matches" className="mb-8">
   280→   280→          <div className="glass-card rounded-3xl border border-gold-500/25 bg-gradient-to-r from-gold-50 via-background to-emerald-50 p-5 dark:from-gold-500/5 dark:via-transparent dark:to-emerald-900/20">
   281→   281→            <div className="mb-4 flex items-center gap-2">
   282→   282→              <span className="flex h-8 w-8 items-center justify-center rounded-full bg-gold-500/15 text-gold-600 dark:text-gold-400">
   283→   283→                <HeartHandshake className="h-4 w-4" aria-hidden />
   284→   284→              </span>
   285→   285→              <h2 className="font-display text-lg font-semibold">Your Matches</h2>
   286→   286→              <Badge className="bg-gold-500/15 text-[11px] text-gold-800 hover:bg-gold-500/15 dark:text-gold-300">
   287→   287→                {mutualProfiles.length} mutual
   288→   288→              </Badge>
   289→   289→            </div>
   290→   290→            <div className="flex gap-4 overflow-x-auto pb-1 scrollbar-thin">
   291→   291→              {mutualProfiles.map((p) => (
   292→   292→                <motion.button
   293→   293→                  key={p.id}
   294→   294→                  whileHover={{ y: -3 }}
   295→   295→                  whileTap={{ scale: 0.97 }}
   296→   296→                  onClick={() => setRevealPair(p)}
   297→   297→                  className="group flex w-36 shrink-0 flex-col items-center gap-2 rounded-2xl border border-gold-500/20 bg-background/70 p-3 text-center shadow-sm transition-colors hover:border-gold-500/45"
   298→   298→                  aria-label={`Open match with ${displayName(p.fullName)}`}
   299→   299→                >
   300→   300→                  <RoundPhoto
   301→   301→                    profile={p}
   302→   302→                    alt={p.fullName}
   303→   303→                    className="h-16 w-16"
   304→   304→                    sizes="64px"
   305→   305→                  />
   306→   306→                  <span className="text-sm font-semibold leading-tight">
   307→   307→                    {displayName(p.fullName)}
   308→   308→                  </span>
   309→   309→                  <span className="flex items-center gap-1 text-[11px] text-gold-700 dark:text-gold-400">
   310→   310→                    <MessageCircle className="h-3 w-3" aria-hidden />
   311→   311→                    Open chat
   312→   312→                  </span>
   313→   313→                </motion.button>
   314→   314→              ))}
   315→   315→            </div>
   316→   316→          </div>
   317→   317→        </section>
   318→   318→      )}
   319→   319→
   320→   320→      {/* ── Tabs ── */}
   321→   321→      <Tabs defaultValue="received">
   322→   322→        <TabsList className="mb-6 flex h-auto w-full flex-wrap justify-start gap-1 rounded-2xl bg-muted/60 p-1.5 sm:inline-flex sm:w-auto">
   323→   323→          <TabsTrigger value="received" className="gap-1.5 rounded-xl px-3 py-2 text-sm">
   324→   324→            <Inbox className="h-4 w-4" aria-hidden />
   325→   325→            Received
   326→   326→            {pendingReceived.length > 0 && (
   327→   327→              <span className="ml-0.5 flex h-5 min-w-5 items-center justify-center rounded-full bg-gold-500 px-1 text-[10px] font-bold text-emerald-950">
   328→   328→                {pendingReceived.length}
   329→   329→              </span>
   330→   330→            )}
   331→   331→          </TabsTrigger>
   332→   332→          <TabsTrigger value="sent" className="gap-1.5 rounded-xl px-3 py-2 text-sm">
   333→   333→            <MailOpen className="h-4 w-4" aria-hidden />
   334→   334→            Sent
   335→   335→            {sent.length > 0 && (
   336→   336→              <span className="text-xs text-muted-foreground">{sent.length}</span>
   337→   337→            )}
   338→   338→          </TabsTrigger>
   339→   339→          <TabsTrigger value="views" className="gap-1.5 rounded-xl px-3 py-2 text-sm">
   340→   340→            <Eye className="h-4 w-4" aria-hidden />
   341→   341→            Who Viewed Me
   342→   342→            {(viewsData?.total ?? 0) > 0 && (
   343→   343→              <span className="text-xs text-muted-foreground">{viewsData?.total}</span>
   344→   344→            )}
   345→   345→          </TabsTrigger>
   346→   346→          <TabsTrigger value="saved" className="gap-1.5 rounded-xl px-3 py-2 text-sm">
   347→   347→            <Gem className="h-4 w-4" aria-hidden />
   348→   348→            Saved
   349→   349→            {savedIds.length > 0 && (
   350→   350→              <span className="text-xs text-muted-foreground">{savedIds.length}</span>
   351→   351→            )}
   352→   352→          </TabsTrigger>
   353→   353→        </TabsList>
   354→   354→
   355→   355→        {/* ═══ RECEIVED ═══ */}
   356→   356→        <TabsContent value="received">
   357→   357→          {isLoading ? (
   358→   358→            <div className="grid gap-5 sm:grid-cols-2 lg:grid-cols-3">
   359→   359→              {[0, 1, 2].map((i) => (
   360→   360→                <Skeleton key={i} className="h-72 rounded-3xl" />
   361→   361→              ))}
   362→   362→            </div>
   363→   363→          ) : received.length === 0 ? (
   364→   364→            <EmptyState
   365→   365→              icon={<Inbox className="h-6 w-6" aria-hidden />}
   366→   366→              title="No proposals yet"
   367→   367→              body="When someone sends a proposal with sincere intentions, it will appear here for you and your family to review."
   368→   368→              action={
   369→   369→                <Button variant="outline" onClick={() => navigate({ name: "browse" })}>
   370→   370→                  Discover Profiles
   371→   371→                </Button>
   372→   372→              }
   373→   373→            />
   374→   374→          ) : (
   375→   375→            <div className="grid gap-5 md:grid-cols-2 lg:grid-cols-3">
   376→   376→              {received.map((interest) => {
   377→   377→                const sender = interest.from as Profile;
   378→   378→                const isMutual = mutualIds.has(sender.id);
   379→   379→                const handled = interest.status !== "pending";
   380→   380→                return (
   381→   381→                  <ProposalCardShell key={interest.id} className="flex flex-col">
   382→   382→                    <div className="flex items-start gap-4">
   383→   383→                      <ArchPhoto
   384→   384→                        profile={sender}
   385→   385→                        alt={sender.fullName}
   386→   386→                        className="h-24 w-20"
   387→   387→                        sizes="80px"
   388→   388→                      />
   389→   389→                      <div className="min-w-0 flex-1">
   390→   390→                        <button
   391→   391→                          onClick={() => onViewProfile(sender.id)}
   392→   392→                          className="block max-w-full truncate text-left font-display text-lg font-semibold leading-snug hover:text-gold-700 dark:hover:text-gold-400"
   393→   393→                        >
   394→   394→                          {displayName(sender.fullName)}
   395→   395→                        </button>
   396→   396→                        <p className="truncate text-sm text-muted-foreground">

   398→   398→                          {sender.country !== "India" ? `, ${sender.country}` : ""}
   399→   399→                        </p>
   400→   400→                        <p className="truncate text-sm text-muted-foreground">{sender.profession}</p>
   401→   401→                        <div className="mt-1.5">
   402→   402→                          <VerificationBadge tier={sender.verificationTier} />
   403→   403→                        </div>
   404→   404→                      </div>
   405→   405→                      <span className="shrink-0 text-[11px] text-muted-foreground">
   406→   406→                        {timeAgo(interest.createdAt)}
   407→   407→                      </span>
   408→   408→                    </div>
   409→   409→
   410→   410→                    {interest.message && (
   411→   411→                      <blockquote className="mt-4 rounded-2xl border-l-2 border-gold-500/60 bg-gold-50/60 px-4 py-3 text-sm italic leading-relaxed text-foreground/85 dark:bg-gold-500/5">

   416→   416→                    <div className="mt-4 flex flex-1 items-end">

   421→   421→                            <Button

   423→   423→                              onClick={() => navigate({ name: "chat", withId: sender.id })}
   424→   424→                              className="gap-1.5 bg-emerald-700 text-white hover:bg-emerald-600"
   425→   425→                            >
   426→   426→                              <MessageCircle className="h-3.5 w-3.5" aria-hidden />
   427→   427→                              Conversation

   429→   429→                          )}
   430→   430→                        </div>
